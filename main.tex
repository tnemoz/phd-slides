    \[\begin{array}{rl}\sup\limits_{\text{\only<5->{\textcolor<5>{red}{Extremal rank-1 }}POVM }\left\{M_x\right\}_x}\visible<-6>{\min\limits_{\substack{\only<-2>{\rho_{ABE}}\\\visible<-6>{\text{PVM }\left\{P_x\right\}_x}}}}&\mathbb{H}(X|E)_{\rho_{XE}}\\\text{s.t.}&\ptr{\only<-6>{B}E}{\only<-2>{\rho_{ABE}}\only<3->{\textcolor<3>{red}{\proj{\psi_{A\only<-6>{B}E}}}}}=\rho_A\\\visible<-6>{&M_x=\ptr{B}{P_x\left(I_A\otimes\ptr{AE}{\only<-2>{\rho_{ABE}}\only<3->{\textcolor<3>{red}{\proj{\psi_{ABE}}}}}\right)}}\end{array}\]
		\begin{itemize}

\begin{frame}{Main result}
	\begin{block}{Theorem}
		Let $\rho_A$ be a $d_A$-dimensional quantum state and $\varepsilon>0$.\pause\ One can extract $\ell$ bits of $\varepsilon$-secure randomness from $\rho_A^{\otimes n}$ whenever

	\end{block}

	\begin{itemize}
		\item Quantifies how useful a given source is for QRNG\pause
		\item Converges to the asymptotic rate $2\log(d_A)-H(A)$
	\end{itemize}

\section{Real-valued pseudorandom states}
%==============================================================================
\begin{frame}{Outline of this talk}
	\vspace{4mm}

\end{frame}

\begin{frame}{Pseudorandom quantum states (PRS)}
	\begin{block}{Definition \citep{JLS18}, informal}
		A keyed family $\left\{\ket{\psi_\key}\right\}_{\key\in\{0,1\}^\lambda}$ of $d$-dimensional states that is\pause
		\begin{itemize}
			\item efficiently preparable from $\key$,\pause
			\item indistinguishable from Haar-random states by any efficient adversary, even given $t=\poly(\lambda)$ copies.
		\end{itemize}
	\end{block}
	\pause
    Used to perform cryptography in Microcrypt.\pause\ Typical security proof, e.g.\ for binary phase states \scite{JLS18,BS19} with $\left\{f_\key\right\}_\key$ a quantum-secure PRF:
	\pause
    \[\underbrace{\left(\frac{1}{\sqrt{d}}\sum_x(-1)^{f_\key(x)}\ket{x}\right)^{\otimes t}}_{\text{PRS}}\;\underset{\text{qPRF}}{\approx_c}\;\underbrace{\left(\frac{1}{\sqrt{d}}\sum_x(-1)^{f(x)}\ket{x}\right)^{\otimes t}}_{f\text{ truly random}}\;\underset{\visible<6->{\textcolor{red}{\varepsilon(t,d)}}}{\approx_s}\;\begin{array}{c}t\text{ copies}\\\text{of a Haar}\\\text{random}\\\text{state}\end{array}\]
	\visible<6->{Advantage $\leqslant\negl(\lambda)+\textcolor{red}{\varepsilon(t,d)}$\visible<6->{: $\varepsilon$ limits the number of copies}}
\end{frame}

\begin{frame}{Real versus complex amplitudes}
	\begin{block}{$\varepsilon$-approximate state $t$-design}
        An ensemble $\left(\mathcal{S},\mu\right)$ such that $\td{\mathbb{E}_{\ket{\psi}\sim\mu}\left[\proj{\psi}^{\otimes t}\right]}{\crho{d}{t}}\leqslant\textcolor<2>{red}{\varepsilon}$, where $\crho{d}{t}\defeq\mathbb{E}_{\ket{\psi}\sim\text{Haar}}\left[\proj{\psi}^{\otimes t}\right]$.
	\end{block}
	\pause\pause
	\begin{center}
		\begin{tabular}{c|c|c}
			& Real amplitudes & Complex amplitudes\\
			\hline
			Best known $\varepsilon$ & $\Theta\left(\frac{t^2}{d}\right)$ & $\Theta\left(\frac{t}{d}\right)$\\
			& \scite{BS19,JMW24,CSBH25} & \scite{BS20,LQSYZ24}
		\end{tabular}
	\end{center}
	\pause
	\begin{center}
		Is $\frac{t^2}{d}$ a fundamental limit for real-valued designs? \visible<5->{\textcolor{red}{\textbf{Yes.}}}
	\end{center}
\end{frame}

\begin{frame}{Step 1: the best design with fixed imaginarity}
	\begin{block}{Proposition}
        Let \[\rrrho{d}{t}\defeq\int_{O\in\mathcal{O}_d}\left(O\proj{0}O^\top\right)^{\otimes t}\,\mathrm{d}\mu(O)\,.\]\pause\ For any ensemble $\mathcal{S}$ of real-valued states, with \[\rho_{\mathcal{S}^t}\defeq\mathbb{E}_{\ket{\psi}\sim\mathcal{S}}\left[\proj{\psi}^{\otimes t}\right]\]
        \pause we have
		\[\left\|\rho_{\mathcal{S}^t}-\crho{d}{t}\right\|\geqslant\left\|\rrrho{d}{t}-\crho{d}{t}\right\|\,.\]
	\end{block}
    \pause\begin{center}$\implies\frac12\left\lVert\rrrho{d}{t}-\crho{d}{t}\right\rVert$ is the best possible $\varepsilon$ for real-valued states\end{center}
\end{frame}

\begin{frame}{Step 2: diagonalizing with symmetries}
	\begin{block}{Schur's lemma, informally}
		If $M$ has symmetries ($VMV^\dagger=M$ for a group of unitaries $V$), its eigenvalues come in \textbf{blocks}: $M$ is constant on each block.\\
		The \textbf{more} symmetries, the \textbf{larger} the blocks.
	\end{block}
	\visible<2->{Both states live on the symmetric subspace $\vee^t\mathbb{C}^d$, spanned by the $\ket{\psi}^{\otimes t}$}\pause
	\begin{columns}[T]
		\begin{column}{.48\textwidth}
			\visible<3->{\centering\textbf{Complex}: $\crho{d}{t}$ commutes with every $U^{\otimes t}$: \textbf{enough} symmetries for a single block\\[1ex]
			\begin{tikzpicture}
				\fill[tptred!30] (0, 0) rectangle (1.4, 1.4);
				\draw[thick] (0, 0) rectangle (1.4, 1.4);
				\node at (0.7, 0.7) {$\lambda$};
			\end{tikzpicture}\\[1ex]
			$\crho{d}{t}=\binom{d+t-1}{t}^{-1}\,\Pi_{\vee^t\mathbb{C}^d}$\par}
		\end{column}
		\begin{column}{.48\textwidth}
			\visible<4->{\centering\textbf{Real}: $\rrrho{d}{t}$ only commutes with the $O^{\otimes t}$: a \textbf{subset} of them, so smaller blocks\\[1ex]
			\begin{tikzpicture}
				\draw[thick] (0, 0) rectangle (1.4, 1.4);
				\fill[blue!20] (0, 0.55) rectangle (0.85, 1.4);
				\fill[blue!40] (0.85, 0.2) rectangle (1.2, 0.55);
				\fill[blue!60] (1.2, 0) rectangle (1.4, 0.2);
				\node at (0.425, 0.975) {$\lambda_0$};
				\node[font=\scriptsize] at (1.025, 0.375) {$\lambda_1$};
			\end{tikzpicture}\\[1ex]
            \visible<5->{\textcolor{red}{Goal: identify the blocks (and the $\lambda_k$)}\par}}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Finding the blocks}
    We're looking for the subspaces that are invariant by applying $O^{\otimes t}$ for all $O\in\mathcal{O}_d$.\pause\ Take $t=2$ and $\ket{\Phi}\defeq\frac{1}{\sqrt{d}}\sum_x\ket{x, x}\in\vee^2\mathbb{C}^d$.\pause\ Then \(O^{\otimes2}\ket{\Phi}=\ket{\Phi}\) for every $O\in\mathcal{O}_d$, but $U^{\otimes2}\ket{\Phi}\neq\ket{\Phi}$ doesn't hold for every $U\in\mathcal{U}_d$.\pause
	\begin{itemize}
		\item $\proj{\Phi}$ is a block on its own for $\rrrho{d}{2}$, but just part of the single block of $\crho{d}{2}$\pause
		\item For larger $t$: many blocks, of many sizes, in dimension $\binom{d+t-1}{t}$\pause
        \item How do we find them?
	\end{itemize}
\end{frame}

\begin{frame}{Our approach: from tensors to polynomials}
	\begin{itemize}\small
        \item Isomorphism $\vee^t\mathbb{C}^d\cong\mathcal{P}_d^t$, homogeneous polynomials of degree $t$ with $d$ variables.\pause\ If $P(X)$ is the image of $\ket{\psi}$, then\pause
            \begin{itemize}
                \item the Bombieri norm \scite{Bea+90} of $P(X)$ is the usual norm of $\ket{\psi}$\pause
                \item the image of $O^{\otimes t}\ket{\psi}$ is $P\left(O^\top X\right)$\pause
            \end{itemize}
		\item Blocks: harmonic decomposition \scite{CW68}, with \emph{harmonic} polynomials $\harm{d}{n}\defeq\left\{H\,:\,\Delta H=0\right\}$
			\[\mathcal{P}_d^t=\harm{d}{t}\oplus\left\|X\right\|^2\harm{d}{t-2}\oplus\left\|X\right\|^4\harm{d}{t-4}\oplus\cdots\]
            with
            \[\|X\|^2=\left(\sum_{i=1}^dX_i\right)^2\,.\]\pause
        \item Eigenvalues:
            \[\only<-8>{\tr{\rrrho{d}{t}\Pi_k}}\only<9>{\underbrace{\tr{\rrrho{d}{t}}}_{\left\lVert\Pi_kP_\delta\right\rVert_{\text{Bombieri}}^2}}\only<10->{\left\lVert\Pi_kP_\delta\right\rVert^2_{\text{Bombieri}}}=\lambda_{d, t, k}\only<6>{\tr{\Pi_k}}\only<7>{\underbrace{\tr{\Pi_k}}_{\dim\left(\mathcal{H}_d^{t-2k}\right)}}\only<8->{\dim\left(\mathcal{H}_d^{t-2k}\right)}\]
	\end{itemize}
    \vfill\visible<11->{\centering$\implies$ We can get a closed-form expression for the eigenvalues!}
\end{frame}

\begin{frame}{Main result: real amplitudes cost a factor $t$}
	\begin{block}{Theorem}
		For $t\lesssim\sqrt{2d\ln d}$:
        \[\td{\rrrho{d}{t}{0}}{\crho{d}{t}}=\frac{t(t-1)}{2d}+\Theta\left(\frac{t^3}{d^2}\right)\,.\]
	\end{block}
	\visible<2->{\centering$\implies$ \textbf{Complex amplitudes are necessary} to beat $\varepsilon=\Theta\left(\frac{t^2}{d}\right)$\par}
\end{frame}

\begin{frame}{From designs to pseudorandom states?}
	\small
	\begin{itemize}
		\item Optimal measurement $\left\{P,I-P\right\}$ distinguishing $\rrrho{d}{t}$ from $\crho{d}{t}$: $P$ projects onto the blocks where $\lambda_{d,t,k}\leqslant\binom{d+t-1}{t}^{-1}$\pause
		\item Works for \emph{any} real ensemble $\mathcal{S}$: $\tr{P\rho_{\mathcal{S}}^{(t)}}=\tr{P\rrrho{d}{t}}$\pause
		\item This bounds $\varepsilon$, not the PRS advantage: is $P$ \emph{efficient}?
	\end{itemize}
	\pause
	\begin{block}{Open problem}
		Can $\left\{P,I-P\right\}$ be implemented to precision $\eta$ in time $\poly\left(t,\log d,\frac{1}{\eta}\right)$?
	\end{block}
	\pause
	\begin{itemize}
		\item Route: Schur transform \scite{Bur+25}, then change of basis from monomials to harmonics\pause
		\item If so (we conjecture it): \textbf{real-valued PRS} are insecure beyond $\sim\sqrt{d}$ copies
	\end{itemize}
\end{frame}

\begin{frame}{Byproducts of the spectral decomposition}
	\begin{itemize}
		\item Exact designs from orthogonal twirling a state with imaginarity following a Beta law\pause
		\item Imaginarity testing requires $\Omega\left(\sqrt{d}\right)$ copies\pause
		\item Optimal measurement for imaginarity estimation, saving copies over \scite{HBK25}\pause
		\item Real-valued state estimation: the Pretty Good Measurement is optimal, with infidelity $\sim\frac{d-1}{4t}$ (numerically) vs.\ $\frac{d-1}{t}$ for complex states
	\end{itemize}
\end{frame}

\begin{frame}{Pseudorandom states meet optics}
	\begin{itemize}
		\item<1-> PRS live in the qudit model: $\ket{\psi}\in\mathbb{C}^d$, given $t$ times\ldots
		\item<2-> \ldots but quantum communication uses light: coherent states
			\[\ket{\alpha}=e^{-\frac{|\alpha|^2}{2}}\sum_{N\geqslant0}\frac{\alpha^N}{\sqrt{N!}}\ket{N},\qquad\text{number of photons }N\sim\operatorname{Poisson}\left(|\alpha|^2\right)\]
        \item<3-> No Haar measure on the set of displacement operators!
	\end{itemize}
	\visible<4->{\centering What is a \emph{uniformly random} coherent state?\par}
    \visible<5->{Mapping of \citet{AL14}: one mode per basis state
            \[\sum_{x=0}^{d-1}\psi_x\ket{x}\;\longmapsto\;\bigotimes_{x=0}^{d-1}\ket{\alpha\,\psi_x}\]}
\end{frame}

\begin{frame}{Haar-random coherent states}
	\small
	Average the mapped state over Haar-random $\ket{\psi}$:
	\[\haarcoh{\alpha}=\sum_{N\geqslant0}\underbrace{e^{-|\alpha|^2}\frac{|\alpha|^{2N}}{N!}}_{\visible<2->{\text{Poisson statistics}}}\;\underbrace{\binom{d+N-1}{N}^{-1}\sum_{n_1+\cdots+n_d=N}\proj{n_1,\ldots,n_d}}_{\visible<3->{N\text{ photons spread uniformly over }d\text{ modes}\;\cong\;\crho{d}{N}}}\]
	\visible<4->{\centering$\implies$ the mean number of photons $|\alpha|^2$ plays the role of the number of copies $t$\par}
	\visible<5->{
	\begin{block}{Pseudorandom coherent states (PRCS)}
		A keyed family of $d$-mode coherent states $\bigotimes_x\ket{\alpha_{\key,x}}$ that is
		\begin{itemize}
			\item normalized: one photon on average,
			\item efficiently preparable from $\key$,
			\item pseudorandom: $\bigotimes_x\ket{\sqrt{t}\,\alpha_{\key,x}}\approx_c\haarcoh{\sqrt{t}}$ for all $t=\poly(\lambda)$.
		\end{itemize}
	\end{block}}
\end{frame}

\begin{frame}{A PRCS construction}
	\begin{block}{Theorem}
		If $\left\{f_\key\right\}_\key$ is a quantum-secure PRF from $\left[2^\lambda\right]$ to $[p]$, then $\displaystyle\left\{\bigotimes_{x=0}^{2^\lambda-1}\ket{\frac{e^{\frac{2\mathrm{i}\pi f_\key(x)}{p}}}{\sqrt{2^\lambda}}}\right\}_\key$ is a PRCS.
	\end{block}
	\pause
	\begin{center}
	\begin{tikzpicture}[st/.style={draw, rounded corners, thick, align=center, minimum height=1cm, font=\scriptsize}]

		\node[st] (d) at (11.5, 0) {$\haarcoh{\sqrt{t}}$};
		\draw[<->, thick] (a) -- (b) node[midway, above, font=\tiny]{qPRF} node[midway, below=.6cm, font=\scriptsize]{$\negl(\lambda)$};
		\visible<3->{\draw[<->, thick] (b) -- (c) node[midway, above, font=\tiny]{random} node[midway, below=.6cm, font=\scriptsize]{$\sqrt{\frac{2d}{p!}\left(\frac td\right)^p}$};}
		\visible<4->{\draw[<->, thick] (c) -- (d) node[midway, above, font=\tiny]{phases} node[midway, below=.6cm, font=\scriptsize]{$\frac{t^2}{d-1}$};}

	\end{center}

	\begin{itemize}
		\item Each mode prepared independently: constant depth\ldots\pause
        \item \ldots but $2^\lambda$ modes: practical limitation

\begin{frame}{What does the optimal POVM for qubits look like?}
	For $\rho_A=\lambda_0\proj{0}+(1-\lambda_0)\proj{1}$ with $\lambda_0\geqslant\frac12$, the rank-1 POVM $\left\{\proj{\varphi_i}\right\}_{i=0}^3$ with
    \begin{equation*}
        \begin{aligned}
            \ket{\varphi_0} &= \frac{1}{2\sqrt{\lambda_0}}\,\ket{0}\\
            \ket{\varphi_{1+k}} &= \sqrt{\frac{4\lambda_0-1}{12\lambda_0}}\,\ket{0}+\frac{e^{\frac{2\mathrm{i}k\pi}{3}}}{\sqrt{3}}\,\ket{1}\qquad k\in\{0,1,2\}
        \end{aligned}
    \end{equation*}
	is extremal and yields a uniform distribution when measuring $\rho_A$.

	\vspace{2mm}
	General dimension: an explicit inductive construction peels off one eigenvalue at a time (thesis, Section~3.4.4).
\end{frame}

\begin{frame}{Is having the POVM rank-1 necessary for independence with the device's inner system?}
    We proved that $\left\{M_i\right\}_i$ \textcolor<2>{red}{rank-1} extremal POVM $\implies\rho_{XE}=f\left(\rho_{AE}\right)\perp\!\!\perp B$.

    \visible<3->{\cite[Theorem~1]{DCZM23}: extremality is enough, in a model without $A$--$B$ entanglement.}

    \visible<4->{Generalization removing the rank-1 assumption: Senno, Davino and Mugnai (private communication, unpublished).}

\end{frame}

\begin{frame}{Where is the intrinsic randomness continuous?}
	\begin{block}{Theorem (rank-1 extremal POVMs)}
		For a continuous conditional entropy $\mathbb{H}$ and $N\leqslant d_A^2$, $M\mapsto\mathcal{I}_{\mathbb{H}}\left(\rho_A,M\right)$ is continuous on rank-1 extremal POVMs with $N$ outcomes: $\rho_{XE}$ only depends on $M$ and a purification of $\rho_A$.
	\end{block}
	\begin{block}{Theorem (full-rank POVMs)}
		If $M\mapsto2^{\mathcal{I}_{\mathbb{H}}\left(\rho_A,M\right)}$ is convex (true for sandwiched Rényi entropies), then $M\mapsto\mathcal{I}_{\mathbb{H}}\left(\rho_A,M\right)$ is continuous on $\left\{M\,\middle|\,\forall x, M_x\succ0\right\}$, the relative interior \scite{Roc70}.
	\end{block}
	\begin{itemize}
		\item Any implemented (noisy) POVM is full-rank\ldots
		\item \ldots but can the optimal value be approached inside this set?
	\end{itemize}
\end{frame}

\begin{frame}{Computing the optimal $\alpha$}
	Minimize over $\alpha\in(1,2]$:
	\[f(\alpha)=H_{\frac{\alpha}{2\alpha-1}}(A)-\frac{\alpha}{n(\alpha-1)}\log(\varepsilon)\]
	\begin{itemize}
		\item With $p=\operatorname{spec}(\rho_A)$ and $q(\beta)_x\propto p_x^\beta$: $f'(\alpha)=\frac{D_{\mathrm{KL}}\left(q\left(\frac{\alpha}{2\alpha-1}\right)\middle\|p\right)+\frac{\log\varepsilon}{n}}{(\alpha-1)^2}$
		\item $\alpha\mapsto D_{\mathrm{KL}}\left(q\left(\frac{\alpha}{2\alpha-1}\right)\middle\|p\right)$ is strictly increasing unless $p$ is uniform on its support
		\item $\implies f$ is unimodal: bisection gives $\alpha^*$ to precision $\delta$ in $\left\lceil\log_2\frac1\delta\right\rceil-1$ steps
		\item As $n\to\infty$: $\alpha=1+\frac{1}{\sqrt{n}}$ recovers the asymptotic rate
	\end{itemize}
\end{frame}

\begin{frame}{Eigenvalues of $\rrho{d}{t}{\delta}$}
	For $d=2$, with $T_n$ the Chebyshev polynomials of the first kind:
	\[\lambda_{2,t,k}(\delta)=\frac{t!\,(1-\delta)^{\frac t2}\,T_{t-2k}\left(\frac{1}{\sqrt{1-\delta}}\right)}{(2t-2k)!!\,(2k)!!}\]
	For $d\geqslant3$, with $C_n^{(\mu)}$ the Gegenbauer polynomials:
	\[\lambda_{d,t,k}(\delta)=\frac{t!\,(t-2k)!\,(d-4)!!\,(d-2)!\,(1-\delta)^{\frac t2}\,C_{t-2k}^{\left(\frac d2-1\right)}\left(\frac{1}{\sqrt{1-\delta}}\right)}{(2t+d-2k-2)!!\,(2k)!!\,(d+t-2k-3)!}\]
	\begin{itemize}
		\item Proof: Bombieri norm of the projection of $P^t$ onto each $\left\|X\right\|^{2k}\harm{d}{t-2k}$, with $P=\alpha X_0+\mathrm{i}\beta X_1$
		\item $\delta=1$: $\rrho{d}{t}{1}$ is maximally mixed on $\harm{d}{t}$
	\end{itemize}
\end{frame}

\begin{frame}{Imaginarity testing}
	A $\left(\beta_0,\beta_1,t\right)$-tester decides whether $\Imaginarity\leqslant\beta_0$ or $\Imaginarity\geqslant\beta_1$ from $t$ copies.
	\begin{block}{Theorem}
		\begin{itemize}
			\item If $\beta_1<\left(\frac13\right)^{\frac{2}{d-1}}$, a tester needs $t\sim\sqrt{2d\ln\left(\frac{1}{\frac23+\beta_1^{\frac{d-1}{2}}}\right)}$ copies.
			\item If $\beta_0>\left(\frac23\right)^{\frac{2}{d-1}}$, a tester needs $t\sim d\,\frac{\beta_0^{\frac{d-1}{2}}-\frac23+\sqrt{\beta_0^{\frac{d-1}{2}}-\frac23}}{\frac53-\beta_0^{\frac{d-1}{2}}}$ copies.
		\end{itemize}
	\end{block}
	Proof: by the exact-design decomposition, the twirl restricted to imaginarities $\geqslant\beta_1$ is $\beta_1^{\frac{d-1}{2}}$-close to $\crho{d}{t}$, so a tester forces $\td{\rrho{d}{t}{0}}{\crho{d}{t}}\geqslant\frac13-\beta_1^{\frac{d-1}{2}}$; conclude with the asymptotics of the main result.
\end{frame}

\begin{frame}{Key rate ingredients}
	Alice uses $2^n$ threshold detectors (efficiency $\eta_{\text{det}}$, dark counts $p_{\text{dark}}$), transmittance $T$:
	\[1-p_{\text{abort}}=\left(1-p_{\text{dark}}\right)^{2^n-1}\left[2^np_{\text{dark}}-\eta_{\text{det}}\left(1-e^{-T|\beta|^2}\right)\left(2^np_{\text{dark}}-1\right)\right]\]
	\[I(A:B)=n+\Pr[A=B]\log\Pr[A=B]+\left(1-\Pr[A=B]\right)\log\frac{1-\Pr[A=B]}{2^n-1}\]
	\[p_{\text{guess}}\leqslant\sqrt{\frac{e^{2|\alpha||\beta|}+\left(|\alpha|^2+|\beta|^2\right)^2}{2^n}}+\frac{|\alpha|^2+|\beta|^2}{2^{\frac n2}}+\Theta\left(\frac{\left(|\alpha|^2+|\beta|^2\right)^2}{2^n}\right)\]
	Pessimistic: Eve gets a copy of both the public key and the ciphertext.
\end{frame}

\begin{frame}{Comparison with other key distribution methods}
	\begin{description}
		\item[Classical PKE] needs \emph{trapdoor} one-way functions (broken by Shor); ours only quantum-secure PRFs, but needs dedicated hardware
		\item[Classical SKE] needs a pre-shared secret; so do we in practice, to authenticate the quantum channel $\implies$ currently the better option
		\item[QKD] everlasting security, but one key per quantum channel; ours: computational security, one broadcast for many users
		\item[Other QPKE] qudit-based \scite{Bar+23,Col23,KTNY24,MW24}; \scite{KTNY24,MW24} resist tampered public keys, unlike ours
	\end{description}
\end{frame}

\begin{frame}{Forrelation in the QCT model}
	$k$-Forrelation: decide whether $\bra{0}HZ_{f_1}H\cdots HZ_{f_k}H\ket{0}$ is close to $0$ \scite{Aar10,AA15}.
	\begin{itemize}
		\item One-way protocol: Alice sends $\Zf{f_1}HZ_{g_2}Z_{f_2}H\cdots HZ_{g_k}Z_{f_k}H\ket{0}$ and $\operatorname{Enc}_{\key_{\text{weak}}}\left(g_1\right)$
		\item Bob computes the overlap with $Z_{g_1}H\ket{0}$ and decides the secret bit
		\item Hope: Eve needs $d^{1-\frac1k}$ classical bits, but each copy gives her only $\log d$ bits
	\end{itemize}
	\pause
	\textcolor{red}{Issue} (found by Jeanne Lucas): Eve's information is about the \emph{state}, not the functions; once $g_1$ leaks, computing the overlap may be much easier $\implies$ the reduction does not hold.
\end{frame}
